var arrivalsUrl = "https://brokenbowvacationcabins.trackhs.com/api/pms/reservations?sortColumn=checkin&sortDirection=asc&arrivalStart="+date+"&arrivalEnd="+date+"&scroll=1&status=confirmed&size=100";
  var arrivalResponse = UrlFetchApp.fetch(arrivalsUrl, options);
var arrivalData = JSON.parse(arrivalResponse.getContentText());
var arrivals = arrivalData._embedded.reservations;
var arrivalCount = arrivals.length;

for (x = 0; x < arrivals.length; x++) {
  //console.log(arrivals[x].nights);
  var feesUrl = "https://brokenbowvacationcabins.trackhs.com/api/pms/reservations/"+arrivals[x].id+"/fees";
  var feesResponse = UrlFetchApp.fetch(feesUrl, options);
  var feesData = JSON.parse(feesResponse.getContentText());
  var fees = feesData._embedded.fees;
  if (units.flat().find(u => u.id === arrivals[x].unitId)) {
    units.flat().find(u => u.id === arrivals[x].unitId).isArrival = "Arrival";
    units.flat().find(u => u.id === arrivals[x].unitId).maintPrep = "Maintenance";
    units.flat().find(u => u.id === arrivals[x].unitId).resId = arrivals[x].id;
  }
  for(i = 0; i < fees.length; i++) {
      if (woodIds.includes(fees[i].chargeId)) {
        if (fees[i].chargeId != 34) {
          units.flat().find(r => r.resId === arrivals[x].id).wood = fees[i].chargeInline.name.match(/[0-9!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]+/g)[0];
        } else {
          units.flat().find(r => r.resId === arrivals[x].id).wood = parseInt(fees[i].quantity) + " Bundle";
        }
        
      } else if (fees[i].chargeId === os) {
        units.flat().find(r => r.resId === arrivals[x].id).hkPrep = "HK";
        units.flat().find(r => r.resId === arrivals[x].id).os = "OS";
      } else if (fees[i].chargeId === romance) {
        units.flat().find(r => r.resId === arrivals[x].id).hkPrep = "HK";
        units.flat().find(r => r.resId === arrivals[x].id).romance = "❤️";
      } 
    }
  
}

var outsUrl = "https://brokenbowvacationcabins.trackhs.com/api/pms/reservations?sortColumn=name&sortDirection=asc&departureStart="+date+"&departureEnd="+date+"&scroll=1&status=Checked%20In&size=100";
var outResponse = UrlFetchApp.fetch(outsUrl, options);
var outsData = JSON.parse(outResponse.getContentText());
var outs = outsData._embedded.reservations;
for (x = 0; x < outs.length; x++) {
  //console.log(outs[x]._embedded.unit.name);
  if (units.flat().find(u => u.id === outs[x].unitId)) {
    units.flat().find(u => u.id === outs[x].unitId).hkPrep = "HK";
    units.flat().find(u => u.id === outs[x].unitId).isCheckout = "Checkout";
  }
}

      detail.getRange(i+2, 5).setValue("TRUE");
      detail.getRange(i+2, 6).setValue("OCCUPIED");

      detail.getRange(i+2, 6).setValue("Vacant");
